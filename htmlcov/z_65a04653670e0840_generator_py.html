3# Synthetic flight generator (SYN1). 

4# 

5"""Profile + CIFP -> a FIX bus stream. 

6 

7:func:`generate` flies the scenario and returns a :class:`Run`: one row per 

8tick of the keys a GPS/AHRS/air-data box would publish (``LAT LONG ALT GS TRACK 

9HEAD TAS IAS VS PITCH ROLL MAGVAR GPS_FIX_TYPE GPS_ACCURACY_HORIZ``, less what a 

10GPS event withholds: see :data:`GPS_EVENTS`), the reference keys 

11(``REFCRS REFXTK REFCDI REFDIS REFWPT REFPT``), and the pilot's inputs to the box 

12under test: the route block once at the start, ``FPLCMD`` to activate the 

13leg into the start fix, and ``FPLCMD RESUME`` when the pilot goes missed. 

14 

15:func:`write_recorder_file` writes a run in the ``data_recorder`` JSON-lines 

16format (``{"frequency", "starttime"}`` header, then one key-to-value row per 

17tick, values as ``[value, annunciate, old, bad, fail, secfail]``), which 

18``data_playback`` replays unchanged. :mod:`fixgw.sim.live` (SYN2) streams 

19``Run.rows`` into a running gateway over netfix instead. 

20"""

36REF_KEYS = ("REFCRS", "REFXTK", "REFCDI", "REFDIS", "REFWPT", "REFPT") 

37SENSOR_KEYS = ("MAGVAR", "GPS_FIX_TYPE", "GPS_ACCURACY_HORIZ", "ALT", "GS", "TAS", "IAS", "VS", 

38 "TRACK", "HEAD", "PITCH", "ROLL", "LAT", "LONG") 

39ESTABLISHED_NM = 0.05

41# What each GPS event stops publishing, and the fix type written meanwhile 

42# (None: GPS_FIX_TYPE is not written at all, so it keeps its last value). 

43# gps_loss_s the receiver has no fix: what the gpsd plugin publishes 

44# below a 2D fix, GPS_FIX_TYPE 1 and nothing needing a fix 

45# gps_link_loss_s the receiver or its link goes dead: nothing at all, so 

46# the last fix type (3) stands 

47# gps_status_loss_s only the receiver's status stream stops while position 

48# keeps coming: the mavlink plugin takes GPS_FIX_TYPE and 

49# GPS_ACCURACY_HORIZ from GPS_RAW_INT but LAT/LONG from 

50# GLOBAL_POSITION_INT, the autopilot's fused estimate 

51# gps_position_loss_s the other way round: LAT/LONG stop while the status 

52# stream still says 3D 

53_GPS_POSITION = ("LAT", "LONG", "TRACK", "GS", "GPS_ACCURACY_HORIZ") 

54GPS_EVENTS = { 

55 "gps_loss_s": (_GPS_POSITION, 1), 

56 "gps_link_loss_s": (_GPS_POSITION + ("GPS_FIX_TYPE",), None), 

57 "gps_status_loss_s": (("GPS_FIX_TYPE", "GPS_ACCURACY_HORIZ"), None), 

58 "gps_position_loss_s": (("LAT", "LONG"), 3), 

59}

62def ias_kt(tas_kt, alt_ft): 

63 """Indicated airspeed for ``tas_kt`` at ``alt_ft`` on a standard day: 

64 EAS from the ISA troposphere density ratio, taking IAS = CAS = EAS (no 

65 position error, no compressibility at these speeds). The air-data box 

66 reports IAS; the profile flies TAS, so this is what it would read.""" 

67 sigma = (1.0 - 6.8755856e-6 * max(alt_ft, 0.0)) ** 4.2558797 

68 return tas_kt * sigma ** 0.5

80def _alt_constraints(cifp, proc): 

81 out = {} 

82 for r in cifp.approach_legs(proc["airport"], proc["approach"]): 

83 if r.fix and r.alt1 and r.path_term != "HM": 

84 out.setdefault(r.fix, float(r.alt1)) 

85 return out

88def generate(profile, frequency_ms=200, cifp_path=None, fplseq=1, gps_accuracy_ft=30.0, cmd_seq_base=0): 

89 if isinstance(profile, str): 

90 profile = load_profile(profile, cifp_override=cifp_path) 

91 cifp = Cifp(cifp_path or profile.cifp_file) 

92 if cifp.cycle != profile.cifp_cycle: 

93 raise GeneratorError(f"profile pins CIFP cycle {profile.cifp_cycle}, file {cifp.path} is {cifp.cycle}") 

94 

95 proc = profile.procedure 

96 mv = profile.magvar_deg 

97 try: 

98 legs = refnav.legs_from_route(cifp, profile) 

99 slots, segments, name, aprtype = fplroute.route_slots(cifp, profile) 

100 except (ValueError, refnav.ReferenceError_) as e: 

101 raise GeneratorError(f"{profile.name}: {e}") 

102 apt = cifp.fix(profile.destination, section="PA") 

103 

104 route_ids = [f.ident for f, _ in cifp.expand_route(profile.enroute)] 

105 end_fix = profile.end.get("after_fix") 

106 if end_fix is not None and end_fix not in route_ids: 

107 raise GeneratorError(f"end.after_fix {end_fix} is not an en-route fix of the route {route_ids}") 

108 if "runway" in profile.start: 

109 # over the threshold, on the climb-out's CA (the leg after the runway) 

110 k = next(i for i, lg in enumerate(legs) if lg.pt == "CA") 

111 act_slot = next(i for i, s in enumerate(slots, start=1) if s["PT"] == "CA") 

112 rwy, _, elev_ft = cifp.runway(profile.departure, profile.start["runway"]) 

113 lat, lon, azi = rwy.lat, rwy.lon, legs[k].course_true 

114 if not elev_ft <= float(profile.start["alt_ft"]) < legs[k].alt_ft: 

115 raise GeneratorError(f"start.alt_ft {profile.start['alt_ft']} is not between the threshold " 

116 f"elevation {elev_ft:.0f} and the climb-out altitude {legs[k].alt_ft:.0f}") 

117 else: 

118 start_fix = profile.start["fix"] 

119 try: 

120 k = next(i for i, lg in enumerate(legs) if lg.ident == start_fix and i > 0) 

121 act_slot = next(i for i, s in enumerate(slots, start=1) if s["ID"] == start_fix and i > 1) 

122 except StopIteration: 

123 raise GeneratorError(f"start fix {start_fix} is not a fix after the first in the route") 

124 prev = legs[k - 1] 

125 to = legs[k] 

126 if to.pt == "DF" or prev.lat is None: 

127 raise GeneratorError(f"start fix {start_fix}: the leg into it is a {to.pt} from where the climb-out " 

128 "ends; start on the runway instead") 

129 line = refnav.GeoLine.between(prev.lat, prev.lon, to.lat, to.lon) 

130 before = float(profile.start.get("before_nm", 0.0)) 

131 lat, lon, azi = line.position(max(0.0, line.length_nm - before)) 

132 

133 a = profile.aircraft 

134 perf = Performance(tas_kt=a.get("tas_kt", 110.0), climb_fpm=a.get("climb_fpm", 700.0), 

135 descent_fpm=a.get("descent_fpm", 600.0), bank_max_deg=a.get("bank_max_deg", 25.0), 

136 turn_rate_deg_s=a.get("turn_rate_deg_s", 3.0)) 

137 ac = Aircraft(perf, State(lat, lon, float(profile.start["alt_ft"]), azi), 

138 profile.wind_from_deg, profile.wind_kt) 

139 dep = cifp.fix(profile.departure, section="PA") if profile.departure else None 

140 ref = refnav.ReferenceNavigator(legs, mv, (apt.lat, apt.lon), k, perf.tas_kt, 

141 perf.turn_rate_deg_s, perf.bank_max_deg, 

142 departure=(dep.lat, dep.lon) if dep else None) 

143 alts = _alt_constraints(cifp, proc) if proc else {} 

144 missed_top = max([alts.get(lg.ident, 0.0) for lg in legs if lg.first_missed or ref_in_missed(legs, lg)] + [0.0]) 

145 

146 dt = frequency_ms / 1000.0 

147 rows = [] 

148 timeline = [] 

149 cmd_seq = cmd_seq_base # a live gateway has already acked earlier commands 

150 route = fplroute.route_block_writes(slots, segments, name, aprtype, cifp.cycle, fplseq) 

151 

152 events = {e.id: {"e": e, "state": "pending", "t0": None, "est": None} for e in profile.events} 

153 leg_start_t = {k: 0.0} 

154 t = 0.0 

155 t_map = None 

156 resume_after = float(profile.pilot.get("missed_after_map_s", 5.0)) 

157 max_s = float(profile.end.get("max_s", 3600.0)) 

158 end_event = profile.end.get("after_event") 

159 out = None 

160 tick = 0 

161 while True: 

162 s = ac.s 

163 n_seq = len(ref.sequences) 

164 out = ref.update(t, s.lat, s.lon, s.alt_ft, s.gs_kt, s.track_true) 

165 route_done = False 

166 for sq in ref.sequences[n_seq:]: 

167 leg_start_t[sq.leg_index] = t 

168 timeline.append({"t": round(t, 3), "what": "sequence", "from": sq.from_ident, "to": sq.to_ident}) 

169 # as the X-Plane tier ends: once end.after_fix is behind, or, with 

170 # no procedure, once the destination is the TO fix 

171 if sq.from_ident == end_fix or (proc is None and sq.leg_index == len(legs) - 1): 

172 route_done = True 

173 

174 row = {} 

175 cmds = [] 

176 if tick == 1: 

177 cmd_seq += 1 

178 cmds.append(f"{cmd_seq} ACT {act_slot}") 

179 timeline.append({"t": round(t, 3), "what": "pilot", "cmd": cmds[-1]}) 

180 if ref.map_suspended and t_map is None: 

181 t_map = t 

182 timeline.append({"t": round(t, 3), "what": "map", "fix": out.wpt}) 

183 if t_map is not None and ref.map_suspended and t >= t_map + resume_after: 

184 if not proc.get("missed", False): 

185 break 

186 ref.go_missed(t, (s.lat, s.lon), s.track_true, s.gs_kt) 

187 leg_start_t[ref.i] = t 

188 cmd_seq += 1 

189 cmds.append(f"{cmd_seq} RESUME") 

190 timeline.append({"t": round(t, 3), "what": "pilot", "cmd": cmds[-1]}) 

191 out = ref.update(t, s.lat, s.lon, s.alt_ft, s.gs_kt, s.track_true) 

192 

193 done_now = _run_events(events, ref, out, t, leg_start_t, ac, timeline) 

194 withheld, fix_type = set(), 3 

195 for ev in events.values(): 

196 for kind in GPS_EVENTS.keys() & ev["e"].do.keys(): 

197 if ev["state"] == "active": 

198 withheld.update(GPS_EVENTS[kind][0]) 

199 # not written beats "no fix", which beats 3D 

200 ft = GPS_EVENTS[kind][1] 

201 fix_type = None if ft is None or fix_type is None else min(fix_type, ft) 

202 

203 # sensor keys first, LAT/LONG last: the engine computes on a position write. 

204 sensors = { 

205 "MAGVAR": mv, 

206 "GPS_FIX_TYPE": fix_type, 

207 "GPS_ACCURACY_HORIZ": gps_accuracy_ft, 

208 "ALT": round(s.alt_ft, 2), 

209 "GS": round(s.gs_kt, 3), 

210 "TAS": round(perf.tas_kt, 3), 

211 "IAS": round(ias_kt(perf.tas_kt, s.alt_ft), 3), 

212 "VS": round(s.vs_fpm, 1), 

213 "TRACK": round(refnav.wrap360(s.track_true), 3), 

214 "HEAD": round(refnav.wrap360(s.heading_true + mv), 3), 

215 "PITCH": round(s.pitch_deg, 2), 

216 "ROLL": round(s.roll_deg, 2), 

217 "LAT": round(s.lat, 8), 

218 "LONG": round(s.lon, 8), 

219 } 

220 for key, value in sensors.items(): 

221 if key not in withheld and value is not None: 

222 row[key] = value 

223 row["REFCRS"] = round(out.crs_mag, 3) 

224 row["REFXTK"] = round(out.xtk_nm, 5) 

225 row["REFCDI"] = round(out.cdi, 5) 

226 row["REFDIS"] = round(out.dis_nm, 4) 

227 row["REFWPT"] = out.wpt 

228 row["REFPT"] = out.pt 

229 if tick == 0: 

230 for key, value in route: 

231 row[key] = value 

232 for c in cmds: 

233 row["FPLCMD"] = c 

234 rows.append((round(t, 3), row)) 

235 

236 if (end_event and end_event in done_now) or route_done: 

237 break 

238 if t > max_s: 238 ↛ 239line 238 didn't jump to line 239 because the condition on line 238 was never true

239 pending = [i for i, ev in events.items() if ev["state"] != "done"] 

240 raise GeneratorError(f"{profile.name}: still running at {max_s:.0f} s; events not done: {pending}") 

241 

242 # vertical: descend to the TO fix's constraint on the approach, climb to the top of the missed, 

243 # and on any other CA (a climb-out) climb to its altitude, which is what ends the leg 

244 if ref.missed: 

245 target = missed_top 

246 vs_lim = None 

247 elif out.pt == "CA": 

248 target = max(s.alt_ft, ref.legs[ref.i].alt_ft) 

249 vs_lim = None 

250 elif ref.map_suspended: 

251 target = s.alt_ft 

252 vs_lim = None 

253 else: 

254 target = min(s.alt_ft, alts.get(out.wpt, s.alt_ft)) 

255 vs_lim = perf.descent_fpm 

256 if out.dis_nm > 0 and s.gs_kt > 0 and s.alt_ft > target: 

257 need = (s.alt_ft - target) / (out.dis_nm / s.gs_kt * 60.0) 

258 vs_lim = min(perf.max_descent_fpm, max(perf.descent_fpm, need)) 

259 ac.step(dt, out.dtk_true, out.xtk_nm, target, out.turn_hint, vs_lim) 

260 t += dt 

261 tick += 1 

262 

263 pending = [i for i, ev in events.items() if ev["state"] != "done"] 

264 if pending and end_event and not route_done: 264 ↛ 265line 264 didn't jump to line 265 because the condition on line 264 was never true

265 raise GeneratorError(f"{profile.name}: run ended with events not done: {pending}") 

266 return Run(profile, frequency_ms, rows, timeline, [s.__dict__ for s in ref.sequences])

269def ref_in_missed(legs, leg): 

270 seen = False 

271 for lg in legs: 271 ↛ 276line 271 didn't jump to line 276 because the loop on line 271 didn't complete

272 if lg.first_missed: 

273 seen = True 

274 if lg is leg: 

275 return seen 

276 return False

279def _run_events(events, ref, out, t, leg_start_t, ac, timeline): 

280 done_now = set() 

281 for eid, ev in events.items(): 

282 e = ev["e"] 

283 if ev["state"] == "pending": 

284 at = e.at 

285 fire = False 

286 if "leg" in at: 

287 frm, _, to = at["leg"].partition("-") 

288 i = ref.i 

289 prev = ref._prev_fix(i) 

290 if ref.legs[i].ident == to and prev is not None and prev.ident == frm and ref._hold is None: 

291 fire = t - leg_start_t.get(i, t) >= float(at.get("after_s", 0.0)) 

292 elif "after" in at: 

293 fire = events[at["after"]]["state"] == "done" 

294 elif "hold_at" in at: 294 ↛ 296line 294 didn't jump to line 296 because the condition on line 294 was always true

295 fire = ref._hold is not None and ref.legs[ref.i].ident == at["hold_at"] 

296 if fire: 

297 ev["state"], ev["t0"] = "active", t 

298 if "offset_nm" in e.do: 

299 ac.offset_nm = float(e.do["offset_nm"]) 

300 timeline.append({"t": round(t, 3), "what": "event", "id": eid, "do": e.do}) 

301 if ev["state"] == "active": 

302 finished = False 

303 if "offset_nm" in e.do: 

304 if ev["est"] is None and abs(out.xtk_nm - ac.offset_nm) < ESTABLISHED_NM: 

305 ev["est"] = t 

306 timeline.append({"t": round(t, 3), "what": "established", "id": eid, 

307 "xtk_nm": round(out.xtk_nm, 4)}) 

308 if ev["est"] is not None and t - ev["est"] >= float(e.do.get("hold_s", 0.0)): 

309 finished = True 

310 elif "hold_min" in e.do: 

311 if ref._hold is None: 311 ↛ 312line 311 didn't jump to line 312 because the condition on line 311 was never true

312 raise GeneratorError(f"event {eid}: hold_min with no hold being flown") 

313 finished = t - ev["t0"] >= float(e.do["hold_min"]) * 60.0 

314 elif GPS_EVENTS.keys() & e.do.keys(): 314 ↛ 317line 314 didn't jump to line 317 because the condition on line 314 was always true

315 kind = next(iter(GPS_EVENTS.keys() & e.do.keys())) 

316 finished = t - ev["t0"] >= float(e.do[kind]) 

317 if finished: 

318 ev["state"] = "done" 

319 done_now.add(eid) 

320 timeline.append({"t": round(t, 3), "what": "done", "id": eid}) 

321 return done_now

324def write_recorder_file(run, path, starttime=None): 

325 """The data_recorder format: header, then every row as 

326 ``key: [value, annunciate, old, bad, fail, secfail]``.""" 

327 starttime = starttime or datetime.datetime(2026, 1, 1, 0, 0, 0) 

328 with open(path, "w") as f: 

329 f.write(json.dumps({"frequency": f"{run.frequency_ms}", "starttime": starttime.isoformat()}, 

330 separators=(",", ":")) + "\n") 

331 for _, row in run.rows: 

332 f.write(json.dumps({k: [v, 0, 0, 0, 0, 0] for k, v in row.items()}, separators=(",", ":")) + "\n")
