5"""Load and validate a scenario profile. Both tiers read the same file, so 

6validation here is about the scenario, not about how either tier flies it: 

7an unknown key is an error (a typo must not silently become "not set"), 

8and nothing X-Plane- or generator-specific is accepted."""

20_TOP = {"name", "description", "cifp", "aircraft", "environment", "start", "route", 

21 "departure", "destination", "pilot", "events", "end", "checks"} 

22_AIRCRAFT = {"tas_kt", "climb_fpm", "descent_fpm", "bank_max_deg", "turn_rate_deg_s"}

70 raise ProfileError(f"{where}: expected a mapping") 

71 extra = set(d) - allowed 

72 if extra: 

73 raise ProfileError(f"{where}: unknown key(s) {sorted(extra)}")

76def load_profile(path, cifp_override=None): 

77 with open(path) as f: 

78 raw = yaml.safe_load(f) 

79 _only(raw, _TOP, "profile") 

80 for k in ("name", "cifp", "start", "route"): 

81 if k not in raw: 81 ↛ 82line 81 didn't jump to line 82 because the condition on line 81 was never true

82 raise ProfileError(f"profile: missing {k!r}") 

83 

84 cifp = raw["cifp"] 

85 _only(cifp, {"cycle", "file"}, "cifp") 

86 cycle = str(cifp.get("cycle", "")) 

87 if not (len(cycle) == 4 and cycle.isdigit()): 

88 raise ProfileError("cifp.cycle: a 4-digit AIRAC cycle is required (pinned, e.g. \"2611\")") 

89 cifp_file = cifp_override or cifp.get("file") 

90 if not cifp_file: 90 ↛ 91line 90 didn't jump to line 91 because the condition on line 90 was never true

91 raise ProfileError("cifp.file: no CIFP file named, and none given on the command line") 

92 if not os.path.isabs(cifp_file) and not cifp_override: 

93 cifp_file = os.path.join(os.path.dirname(os.path.abspath(path)), cifp_file) 

94 

95 aircraft = raw.get("aircraft", {}) or {} 

96 _only(aircraft, _AIRCRAFT, "aircraft") 

97 env = raw.get("environment", {}) or {} 

98 _only(env, _ENV, "environment") 

99 if "magvar_deg" not in env: 99 ↛ 100line 99 didn't jump to line 100 because the condition on line 99 was never true

100 raise ProfileError("environment.magvar_deg: required (FIX MAGVAR, positive west)") 

101 wind = env.get("wind", {}) or {} 

102 _only(wind, {"from_deg", "kt"}, "environment.wind") 

103 

104 start = raw["start"] 

105 _only(start, _START, "start") 

106 if ("fix" in start) == ("runway" in start) or "alt_ft" not in start: 

107 raise ProfileError("start: alt_ft and exactly one of fix / runway are required") 

108 if "runway" in start and "before_nm" in start: 

109 raise ProfileError("start: before_nm goes with fix; a runway start is over the threshold") 

110 if start.get("heading", "on_course") != "on_course": 110 ↛ 111line 110 didn't jump to line 111 because the condition on line 110 was never true

111 raise ProfileError("start.heading: only on_course is supported") 

112 

113 route = raw["route"] 

114 if not isinstance(route, list) or not route: 

115 raise ProfileError("route: a list of waypoint / airway items, then at most one procedure") 

116 enroute, proc, climb = [], None, None 

117 for n, item in enumerate(route): 

118 where = f"route[{n}]" 

119 if not isinstance(item, dict) or len(item.keys() & set(_ITEMS)) != 1: 

120 raise ProfileError(f"{where}: exactly one of {' / '.join(_ITEMS)}") 

121 _only(item, set(_ITEMS), where) 

122 if "climb_out" in item: 

123 if n != 0: 

124 raise ProfileError(f"{where}: climb_out opens the route; it must be the first item") 

125 climb = item["climb_out"] 

126 _only(climb, _CLIMB_OUT, f"{where}.climb_out") 

127 if not isinstance(climb.get("runway"), str) or not isinstance(climb.get("alt_ft"), (int, float)) \ 

128 or isinstance(climb.get("alt_ft"), bool): 

129 raise ProfileError(f"{where}.climb_out: {{runway: RW30, alt_ft: 3000}}") 

130 # ARINC 424 lets a CA be followed by a DF but not a TF: the leg 

131 # after it starts wherever the climb ended, not at a fix 

132 if len(route) < 2 or "waypoint" not in route[1]: 

133 raise ProfileError(f"{where}.climb_out: put a waypoint after it (flown direct from " 

134 "where the climb ends)") 

135 climb = {"runway": climb["runway"], "alt_ft": float(climb["alt_ft"])} 

136 elif "procedure" in item: 

137 if n != len(route) - 1: 

138 raise ProfileError(f"{where}: the procedure must be the last route item") 

139 proc = item["procedure"] 

140 _only(proc, _PROC, f"{where}.procedure") 

141 for k in ("airport", "approach", "transition"): 

142 if k not in proc: 142 ↛ 143line 142 didn't jump to line 143 because the condition on line 142 was never true

143 raise ProfileError(f"{where}.procedure: missing {k!r}") 

144 elif "waypoint" in item: 

145 if not isinstance(item["waypoint"], str) or not item["waypoint"]: 145 ↛ 146line 145 didn't jump to line 146 because the condition on line 145 was never true

146 raise ProfileError(f"{where}.waypoint: a fix ident") 

147 enroute.append({"kind": "waypoint", "fix": item["waypoint"]}) 

148 else: 

149 awy = item["airway"] 

150 _only(awy, {"id", "exit"}, f"{where}.airway") 

151 if "id" not in awy or "exit" not in awy: 

152 raise ProfileError(f"{where}.airway: {{id: V27, exit: AVOLS}}") 

153 if not enroute: 

154 raise ProfileError(f"{where}.airway: needs an entry fix: put a waypoint before it") 

155 if enroute[-1]["kind"] == "airway": 

156 raise ProfileError(f"{where}.airway: an airway straight after an airway is not supported " 

157 "(the junction fix would belong to both segments)") 

158 enroute.append({"kind": "airway", "id": str(awy["id"]), "exit": str(awy["exit"])}) 

159 destination = raw.get("destination") 

160 if proc is None and not destination: 

161 raise ProfileError("destination: required when the route has no procedure") 

162 if proc is not None and destination and destination != proc["airport"]: 

163 raise ProfileError(f"destination: {destination} is not the procedure's airport {proc['airport']}") 

164 for k in ("departure", "destination"): 

165 if raw.get(k) is not None and not isinstance(raw[k], str): 165 ↛ 166line 165 didn't jump to line 166 because the condition on line 165 was never true

166 raise ProfileError(f"{k}: an airport ident") 

167 if climb is not None and not raw.get("departure"): 

168 raise ProfileError("departure: required with climb_out (the airport its runway is at)") 

169 if "runway" in start and (climb is None or start["runway"] != climb["runway"]): 

170 raise ProfileError(f"start.runway: {start['runway']} is not the runway of a climb_out opening the route") 

171 

172 pilot = raw.get("pilot", {}) or {} 

173 _only(pilot, _PILOT, "pilot") 

174 

175 events = [] 

176 ids = set() 

177 for n, e in enumerate(raw.get("events", []) or []): 

178 _only(e, {"id", "at", "do"}, f"events[{n}]") 

179 eid = e.get("id") or f"event{n + 1}" 

180 if eid in ids: 180 ↛ 181line 180 didn't jump to line 181 because the condition on line 180 was never true

181 raise ProfileError(f"events[{n}]: duplicate id {eid!r}") 

182 ids.add(eid) 

183 at, do = e.get("at") or {}, e.get("do") or {} 

184 _only(at, {"leg", "after_s", "after", "hold_at"}, f"events[{n}].at") 

185 _only(do, set(_DO) | {"hold_s"}, f"events[{n}].do") 

186 if sum(k in at for k in ("leg", "after", "hold_at")) != 1: 186 ↛ 187line 186 didn't jump to line 187 because the condition on line 186 was never true

187 raise ProfileError(f"events[{n}].at: exactly one of leg / after / hold_at") 

188 if at.get("after") and at["after"] not in ids: 188 ↛ 189line 188 didn't jump to line 189 because the condition on line 188 was never true

189 raise ProfileError(f"events[{n}].at.after: {at['after']!r} is not an earlier event") 

190 if sum(k in do for k in _DO) != 1: 

191 raise ProfileError(f"events[{n}].do: exactly one of {' / '.join(_DO)}") 

192 if "hold_s" in do and "offset_nm" not in do: 

193 raise ProfileError(f"events[{n}].do: hold_s goes with offset_nm") 

194 events.append(Event(eid, at, do)) 

195 

196 end = raw.get("end", {}) or {} 

197 _only(end, _END, "end") 

198 if end.get("after_event") and end["after_event"] not in ids: 198 ↛ 199line 198 didn't jump to line 199 because the condition on line 198 was never true

199 raise ProfileError(f"end.after_event: no event {end['after_event']!r}") 

200 if end.get("after_fix") is not None and not enroute: 

201 # the X-Plane tier counts it on the en-route legs only 

202 raise ProfileError("end.after_fix: names an en-route fix, and the route has none") 

203 

204 checks = raw.get("checks", []) or [] 

205 for n, c in enumerate(checks): 

206 _only(c, _CHECK_KEYS, f"checks[{n}]") 

207 if len(_CHECK_KINDS & set(c)) != 1: 207 ↛ 208line 207 didn't jump to line 208 because the condition on line 207 was never true

208 raise ProfileError(f"checks[{n}]: exactly one of key_pair / sequence / flagged") 

209 if "key_pair" in c and (not isinstance(c["key_pair"], list) or len(c["key_pair"]) != 2): 209 ↛ 210line 209 didn't jump to line 210 because the condition on line 209 was never true

210 raise ProfileError(f"checks[{n}].key_pair: [ENGINE_KEY, REFERENCE_KEY]") 

211 if "flagged" in c: 

212 if not isinstance(c["flagged"], list) or not c["flagged"]: 

213 raise ProfileError(f"checks[{n}].flagged: a list of engine keys") 

214 silent = c.get("while_silent") 

215 if (c.get("while_no_fix") is True) == (silent is not None): 

216 raise ProfileError(f"checks[{n}].flagged: exactly one condition, while_no_fix: true " 

217 "or while_silent: [KEY, ...]") 

218 if silent is not None and (not isinstance(silent, list) or not silent): 

219 raise ProfileError(f"checks[{n}].while_silent: a list of input keys") 

220 if "while_silent" in c and "flagged" not in c: 

221 raise ProfileError(f"checks[{n}].while_silent goes with flagged") 

222 sws = c.get("skip_when_silent") 

223 if sws is not None and (not isinstance(sws, dict) or not sws 

224 or not all(isinstance(v, (int, float)) for v in sws.values())): 

225 raise ProfileError(f"checks[{n}].skip_when_silent: {{KEY: seconds, ...}}") 

226 swd = c.get("skip_when_differ") 

227 if swd is not None and (not isinstance(swd, list) or len(swd) < 2): 

228 raise ProfileError(f"checks[{n}].skip_when_differ: [KEY, KEY, ...]") 

229 mcs = c.get("min_compared_share") 

230 if mcs is not None and ("key_pair" not in c or isinstance(mcs, bool) 

231 or not isinstance(mcs, (int, float)) or not 0.0 < mcs <= 1.0): 

232 raise ProfileError(f"checks[{n}].min_compared_share: a share, 0 < share <= 1, on a key_pair check") 

233 

234 return Profile( 

235 name=raw["name"], path=os.path.abspath(path), description=raw.get("description", ""), 

236 cifp_cycle=cycle, cifp_file=cifp_file, aircraft=aircraft, 

237 magvar_deg=float(env["magvar_deg"]), wind_from_deg=float(wind.get("from_deg", 0.0)), 

238 wind_kt=float(wind.get("kt", 0.0)), start=start, procedure=proc, pilot=pilot,

241 )
