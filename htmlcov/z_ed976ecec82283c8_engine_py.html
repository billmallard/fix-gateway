3# Demo engine (DEMO3, makerplane/briefs/demo_mode_redesign.md section 6.1). 

4# 

5"""A small piston-engine model driven by the flight phase, and the scripted 

6CHT event. 

7 

8The demo's route rows already say what the airplane is doing: ``VS`` climbs 

9and descends, and the reference leg type ``REFPT`` says when it is in the 

10hold. :func:`phases` turns that into a power phase, and :func:`add_engine` 

11steps a first-order model through the rows and adds the engine keys 

12(:data:`ENGINE_KEYS`) once a second. Fast things (RPM, manifold pressure, 

13fuel flow) follow the throttle in seconds, cylinder heads in about a minute, 

14the oil in a few. ``FUELQ1``/``FUELQ2`` burn down by ``FUELF1``. 

15 

16A climb opens at takeoff (go-around) power for :data:`TAKEOFF_S`, then climb 

17power. In the first climb, one cylinder runs hot: its CHT heads for a point 

18past the aircraft's configured ``highAlarm``, so the gauge crosses its warn 

19and then its alarm limit. Once it has sat past the alarm for 

20:data:`EVENT_HOT_S`, the pilot enriches: fuel flow up, EGTs down, and the 

21CHT falls back below the warn limit. The demo writes only the CHT. Whether 

22that is a warning or an alarm is for the configured limits and the shipped 

23``annunciate`` plugin to say, as it would be in flight. 

24 

25Everything here is deterministic: the same rows and limits give the same 

26engine every loop, so a test can read the event off the rows. 

27""" 

28 

29import math 

30 

31ENGINE_KEYS = ("TACH1", "MAP1", "FUELF1", "FUELP1", "OILP1", "OILT1", 

32 "CHT11", "CHT12", "CHT13", "CHT14", "EGT11", "EGT12", "EGT13", "EGT14", 

33 "VOLT", "CURRNT", "FUELQ1", "FUELQ2", "OAT") 

34 

35TAKEOFF, CLIMB, CRUISE, DESCENT, HOLD = "takeoff", "climb", "cruise", "descent", "hold" 

36 

37# Power by phase: RPM, manifold inHg, fuel gal/hr, then the steady EGT, CHT, 

38# oil pressure and oil temperature it settles to, a fixed-pitch 4-cylinder 

39# inside the shipped init_data/default.ini limits. 

40POWER = { 

41 TAKEOFF: dict(TACH1=2550, MAP1=27.5, FUELF1=12.5, EGT=540.0, CHT=192.0, OILP1=78.0, OILT1=88.0), 

42 CLIMB: dict(TACH1=2450, MAP1=25.5, FUELF1=10.5, EGT=570.0, CHT=188.0, OILP1=75.0, OILT1=90.0), 

43 CRUISE: dict(TACH1=2350, MAP1=23.0, FUELF1=8.2, EGT=600.0, CHT=175.0, OILP1=70.0, OILT1=87.0), 

44 HOLD: dict(TACH1=2200, MAP1=20.0, FUELF1=6.8, EGT=580.0, CHT=165.0, OILP1=68.0, OILT1=85.0), 

45 DESCENT: dict(TACH1=2100, MAP1=17.0, FUELF1=5.0, EGT=470.0, CHT=150.0, OILP1=66.0, OILT1=83.0), 

46} 

47# First-order time constants, seconds 

48TAU = dict(TACH1=2.0, MAP1=2.0, FUELF1=2.0, OILP1=4.0, EGT=8.0, CHT=45.0, OILT1=120.0) 

49CHT_OFFSET = (-3.0, 2.0, 5.0, -4.0) # cylinders 1-4 do not read alike 

50EGT_OFFSET = (-10.0, 6.0, 12.0, -8.0) 

51 

52VS_CLIMB_FPM = 300.0 

53VS_DESCENT_FPM = -300.0 

54TAKEOFF_S = 60.0 

55FUEL_START_GAL = 18.0 # each tank, at the top of every loop 

56 

57# The scripted event 

58HOT_CYLINDER = 3 

59EVENT_DELAY_S = 20.0 # into the first climb 

60EVENT_OVER_ALARM = 10.0 # the hot CHT heads for highAlarm + this 

61EVENT_TAU = 25.0 

62EVENT_HOT_S = 20.0 # past the alarm before the pilot enriches 

63ENRICH_FUELF = 1.8 

64ENRICH_EGT = -45.0 

65

67def phases(rows): 

68 """The power phase of each row: hold on an ``HM`` leg, climb or descent 

69 by ``VS``, cruise otherwise. A climb's first :data:`TAKEOFF_S` is takeoff 

70 power; a level-off shorter than 10 s does not end the climb.""" 

71 out = [] 

72 phase = CRUISE 

73 vs = 0.0 

74 climb_t0 = None 

75 left_climb_t = None 

76 for t, row in rows: 

77 vs = row.get("VS", vs) 

78 if row.get("REFPT", "") == "HM": 

79 phase = HOLD 

80 elif vs > VS_CLIMB_FPM: 

81 phase = CLIMB 

82 elif vs < VS_DESCENT_FPM: 

83 phase = DESCENT 

84 else: 

85 phase = CRUISE 

86 if phase == CLIMB: 

87 if climb_t0 is None or (left_climb_t is not None and t - left_climb_t > 10.0): 

88 climb_t0 = t 

89 left_climb_t = None 

90 if t - climb_t0 < TAKEOFF_S: 

91 phase = TAKEOFF 

92 elif climb_t0 is not None and left_climb_t is None: 

93 left_climb_t = t 

94 out.append(phase) 

95 return out

98def event_limits(warn, alarm): 

99 """(warn, alarm) when both are set and alarm is above warn, else None.""" 

100 try: 

101 warn, alarm = float(warn), float(alarm) 

102 except (TypeError, ValueError): 

103 return None 

104 return (warn, alarm) if alarm > warn else None

107def ambient_inhg(alt_ft): 

108 return 29.92 * (1.0 - 6.8756e-6 * alt_ft) ** 5.2559 

109 

110 

111def _jitter(t, period, amp, phase=0.0): 

112 return amp * math.sin(2.0 * math.pi * t / period + phase)

115class Engine: 

116 """The model's state, stepped row by row. ``limits`` is the hot 

117 cylinder's (highWarn, highAlarm) or None for no event.""" 

118 

119 def __init__(self, limits=None, first=CRUISE): 

120 p = POWER[first] 

121 self.v = {k: float(p[k]) for k in ("TACH1", "MAP1", "FUELF1", "OILP1", "OILT1")} 

122 self.cht = [p["CHT"] + o for o in CHT_OFFSET] 

123 self.egt = [p["EGT"] + o for o in EGT_OFFSET] 

124 self.fuel = [FUEL_START_GAL, FUEL_START_GAL] 

125 self.limits = limits 

126 self.event = "pending" if limits else "none" 

127 self.climbs = 0 

128 self.climb_t0 = None 

129 self.hot_since = None 

130 self.rich = False 

131 self.marks = [] # [(t, what)] for the status and the tests 

132 

133 def _lag(self, value, target, tau, dt): 

134 return value + (target - value) * (1.0 - math.exp(-dt / tau)) 

135 

136 def step(self, t, dt, phase, alt_ft): 

137 p = POWER[phase] 

138 climbing = phase in (TAKEOFF, CLIMB) 

139 if climbing and self.climb_t0 is None: 

140 self.climb_t0 = t 

141 self.climbs += 1 

142 elif not climbing: 

143 self.climb_t0 = None 

144 self.rich = False 

145 

146 hot = HOT_CYLINDER - 1 

147 if self.event == "pending" and self.climbs == 1 and climbing and t - self.climb_t0 >= EVENT_DELAY_S: 

148 self.event = "hot" 

149 self.marks.append((t, "hot")) 

150 if self.event == "hot": 

151 if self.cht[hot] > self.limits[1]: 

152 if self.hot_since is None: 

153 self.hot_since = t 

154 elif t - self.hot_since >= EVENT_HOT_S: 

155 self.event = "enriched" 

156 self.rich = True 

157 self.marks.append((t, "enrich")) 

158 elif self.event == "enriched" and self.cht[hot] < self.limits[0]: 

159 self.event = "done" 

160 self.marks.append((t, "recovered")) 

161 

162 rich = self.rich or self.event == "enriched" 

163 targets = dict(p) 

164 targets["MAP1"] = min(p["MAP1"], ambient_inhg(alt_ft) - 0.5) 

165 if rich: 

166 targets["FUELF1"] = p["FUELF1"] + ENRICH_FUELF 

167 for k in ("TACH1", "MAP1", "FUELF1", "OILP1", "OILT1"): 

168 self.v[k] = self._lag(self.v[k], targets[k], TAU[k], dt) 

169 for i in range(4): 

170 target = p["CHT"] + CHT_OFFSET[i] 

171 tau = TAU["CHT"] 

172 if i == hot and self.event == "hot": 

173 target, tau = self.limits[1] + EVENT_OVER_ALARM, EVENT_TAU 

174 elif i == hot and self.event == "enriched": 

175 tau = EVENT_TAU 

176 self.cht[i] = self._lag(self.cht[i], target, tau, dt) 

177 egt = p["EGT"] + EGT_OFFSET[i] + (ENRICH_EGT if rich else 0.0) 

178 self.egt[i] = self._lag(self.egt[i], egt, TAU["EGT"], dt) 

179 burn = self.v["FUELF1"] * dt / 3600.0 / 2.0 

180 self.fuel = [max(0.0, q - burn) for q in self.fuel] 

181 

182 def values(self, t, alt_ft): 

183 """The keys as the sensors would read them at ``t``.""" 

184 out = { 

185 "TACH1": int(round(self.v["TACH1"] + _jitter(t, 7.0, 6.0))), 

186 "MAP1": round(self.v["MAP1"] + _jitter(t, 5.0, 0.05), 2), 

187 "FUELF1": round(self.v["FUELF1"] + _jitter(t, 9.0, 0.05), 2), 

188 "FUELP1": round(40.5 + _jitter(t, 11.0, 0.3), 1), 

189 "OILP1": round(self.v["OILP1"] + _jitter(t, 13.0, 0.4), 1), 

190 "OILT1": round(self.v["OILT1"], 1), 

191 "VOLT": round(14.1 + _jitter(t, 17.0, 0.04), 2), 

192 "CURRNT": round(14.0 + _jitter(t, 23.0, 0.6), 1), 

193 "FUELQ1": round(self.fuel[0], 2), 

194 "FUELQ2": round(self.fuel[1], 2), 

195 "OAT": round(15.0 - 1.98 * alt_ft / 1000.0, 1), 

196 } 

197 for i in range(4): 

198 out["CHT1{}".format(i + 1)] = round(self.cht[i] + _jitter(t, 19.0, 0.3, i), 1) 

199 out["EGT1{}".format(i + 1)] = round(self.egt[i] + _jitter(t, 3.0, 1.5, i), 1) 

200 return out

203def add_engine(rows, limits=None): 

204 """``rows`` with the engine keys added on the first row of each whole 

205 second, and the :class:`Engine` that produced them (its ``marks`` say 

206 when the event ran). ``limits`` as for :class:`Engine`.""" 

207 ph = phases(rows) 

208 eng = Engine(limits, first=ph[0] if ph else CRUISE) 

209 out = [] 

210 alt = 0.0 

211 last_t = None 

212 sec = None 

213 for (t, row), phase in zip(rows, ph): 

214 alt = row.get("ALT", alt) 

215 eng.step(t, 0.0 if last_t is None else t - last_t, phase, alt) 

216 last_t = t 

217 if sec != int(t): 

218 sec = int(t) 

219 row = dict(row) 

220 row.update(eng.values(t, alt)) 

221 out.append((t, row)) 

222 return out, eng
