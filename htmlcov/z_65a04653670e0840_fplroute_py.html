3# Route block encoder for the synthetic flight tier (SYN1). 

4# 

5"""Turns a scenario's route into the FIX route block (``FPLf*``, the RS1 

6segment table, ``FPLCOUNT``/``FPLSEGCNT``/``FPLNAME``/``FPLAPRTYPE``/ 

7``FPLDBCYC``, then ``FPLSEQ`` last) exactly as an editor would publish it 

8(``doc/flightplan_keys.md``). It stands in for the pilot loading the 

9procedure on the display; nothing here reads or imports the engine. 

10 

11Assembling a coded procedure into slots is pyEfis's job (PA7/PA9, 

12``pyefis/flightplan/proc_load.py``); this encoder has to make the same 

13choices, and pyEfis's ``tests/flightplan/test_fplroute_conformance.py`` 

14checks that it does, profile by profile, against a fixture regenerated 

15from this file (``tests/flightplan/fplroute_fixture.py``). Change one here 

16and regenerate that fixture. The choices, so a reviewer can see them: 

17 

18- the transition's last fix and the final's ``IF`` (the same fix, coded 

19 twice) are one slot; 

20- a ``CA`` leg has no fix, so its slot has an empty ident and a position on 

21 its coded course from the previous fix, ``ca_point_nm`` along (the 

22 engine flies a ``CA`` along its course through that point, so any point 

23 on the course line gives the same leg); 

24- ``FPLfALT`` and ``FPLfSPD`` as pyEfis's ``proc_load`` writes them: the 

25 altitude description mapped by :func:`_alt_field`, the record's speed 

26 limit in knots (0 when none); 

27- ``FPLfFLAGS``: fly-over from the description code's second character, 

28 IAF/FAF/MAP from its fourth, MAHP on the slots at the missed approach's 

29 hold fix, and ``0x20`` (from a coded procedure) on every slot; 

30- segment rows: the transition plus final as one approach row (kind 4, 

31 ``RNAV<rwy>.<transition>``), the missed approach as one row (kind 5). 

32 

33En-route items (:func:`route_slots`) follow RS1 as written: a waypoint is 

34one kind-0 row; an airway is one kind-1 row from its entry fix (the 

35waypoint before it) to its exit; the ``departure`` airport, when named, is 

36the first row and the ``destination`` the last when there is no procedure. 

37A procedure's rows follow the en-route ones. Every en-route slot is a 

38``TF`` with no altitude, course or flags: the engine flies a ``TF`` from its 

39two fixes alone, and an airway's minimum altitudes are not encoded yet. 

40 

41A ``climb_out`` (DEMO9) is one departure row (kind 2, ``ID`` the runway) 

42after the departure airport's: the runway threshold slot, then a ``CA`` 

43slot on the runway's magnetic bearing, ``+<alt_ft>``, placed on that course 

44``ca_point_nm`` from the threshold as a procedure's ``CA`` is. It is not a 

45published procedure, so no slot carries ``0x20``. The en-route slot after 

46it is a ``DF``: ARINC 424 does not follow a ``CA`` with a ``TF``, and the 

47engine flies a ``DF`` from wherever the climb ended. 

48"""

68def _alt_field(desc, alt1, alt2): 

69 """The packed ``FPLfALT`` for an ARINC 424 altitude description (col 83) 

70 and its two altitudes, mapped as pyEfis's ``proc_load._alt_text`` maps 

71 them (PA7): the step-down and glide-slope codes keep the constraint on 

72 the fix, so ``V`` (a step-down fix) is ``+``, at or above. A code not 

73 listed is refused rather than guessed.""" 

74 a1 = str(int(alt1)) if alt1 else "" 

75 a2 = str(int(alt2)) if alt2 else "" 

76 if desc in ("+", "H", "J", "V"): 

77 return "+" + a1 if a1 else "" 

78 if desc in ("-", "Y"): 78 ↛ 79line 78 didn't jump to line 79 because the condition on line 78 was never true

79 return "-" + a1 if a1 else "" 

80 if desc in ("", "@", "G", "I", "X"): 80 ↛ 82line 80 didn't jump to line 82 because the condition on line 80 was always true

81 return "@" + a1 if a1 else "" 

82 if desc == "B": 

83 if not (a1 and a2): 

84 return "" 

85 lo, hi = sorted((int(a1), int(a2))) 

86 return f"B{lo},{hi}" 

87 if desc == "C": 

88 return "+" + a2 if a2 else "" 

89 raise ValueError(f"altitude description {desc!r}: not encoded")

92def procedure_slots(cifp, airport, approach, transition, magvar_deg, ca_point_nm=2.0): 

93 """(slots, segments) for ``approach`` via ``transition``. Each slot is a 

94 dict of the 14 ``FPLf*`` fields; each segment a dict of the six 

95 ``FPLS<k>*`` fields.""" 

96 raw = cifp.approach_legs(airport, approach) 

97 trans = [r for r in raw if r.route_type == "A" and r.transition == transition] 

98 final = [r for r in raw if r.route_type != "A"] 

99 slots = [] 

100 kinds = [] 

101 in_missed = False 

102 hold_fix = None 

103 for r in final: 

104 if r.path_term == "HM": 

105 hold_fix = r.fix 

106 for part, records in (("A", trans), ("F", final)): 

107 for r in records: 

108 if r.path_term == "IF" and slots and slots[-1]["ID"] == r.fix: 

109 continue 

110 if r.first_missed: 

111 in_missed = True 

112 slot = {"ID": r.fix, "LAT": 0.0, "LON": 0.0, "TYPE": 4, "PT": r.path_term, 

113 "CRS": 0.0, "DST": 0.0, "ALT": _alt_field(r.alt_desc, r.alt1, r.alt2), 

114 "SPD": int(r.speed or 0), "SGI": 0, "FLAGS": FLAG_FROM_PROCEDURE, 

115 "CTRLAT": 0.0, "CTRLON": 0.0, "TURN": r.turn} 

116 if r.fix: 

117 f = cifp.fix(r.fix, airport=airport if r.fix_section[0] == "P" else None, 

118 section=r.fix_section if r.fix_section.strip() else None) 

119 slot["LAT"], slot["LON"], slot["TYPE"] = f.lat, f.lon, TYPE_BY_KIND[f.kind] 

120 if r.course_mag is not None: 

121 slot["CRS"] = r.course_mag 

122 if r.path_term == "CA": 

123 prev = next(s for s in reversed(slots) if s["ID"]) 

124 true_course = refnav.wrap360(r.course_mag - magvar_deg) 

125 slot["LAT"], slot["LON"] = refnav.direct(prev["LAT"], prev["LON"], true_course, ca_point_nm) 

126 if r.path_term == "HM": 

127 if r.dist_or_time.startswith("T"): 127 ↛ 128line 127 didn't jump to line 128 because the condition on line 127 was never true

128 raise ValueError("time-coded hold: leg length needs a speed, not encoded yet") 

129 slot["DST"] = int(r.dist_or_time) / 10.0 

130 if r.flyover: 

131 slot["FLAGS"] |= FLAG_FLYOVER 

132 if r.is_iaf and part == "A": 

133 slot["FLAGS"] |= FLAG_IAF 

134 if r.is_faf: 

135 slot["FLAGS"] |= FLAG_FAF 

136 if r.is_map: 

137 slot["FLAGS"] |= FLAG_MAP 

138 if in_missed and r.fix and r.fix == hold_fix: 

139 slot["FLAGS"] |= FLAG_MAHP 

140 slots.append(slot) 

141 kinds.append(SEGKIND_MISSED if in_missed else SEGKIND_APPROACH) 

142 segments = [] 

143 for kind in (SEGKIND_APPROACH, SEGKIND_MISSED): 

144 idx = [i for i, k in enumerate(kinds, start=1) if k == kind] 

145 if not idx: 145 ↛ 146line 145 didn't jump to line 146 because the condition on line 145 was never true

146 continue 

147 first, last = idx[0], idx[-1] 

148 rwy = approach[1:] if approach[0] == "R" else approach 

149 ident = f"RNAV{rwy}.{transition}" if kind == SEGKIND_APPROACH else f"RNAV{rwy}" 

150 segments.append({"KIND": kind, "ID": ident, "ENT": slots[first - 1]["ID"], 

151 "EXT": slots[last - 1]["ID"], "FIRST": first, "LAST": last}) 

152 for i in idx: 

153 slots[i - 1]["SGI"] = len(segments) 

154 return slots, segments

157def _fix_slot(f): 

158 return {"ID": f.ident, "LAT": f.lat, "LON": f.lon, "TYPE": TYPE_BY_KIND[f.kind], "PT": "TF", 

159 "CRS": 0.0, "DST": 0.0, "ALT": "", "SPD": 0, "SGI": 0, "FLAGS": 0, 

160 "CTRLAT": 0.0, "CTRLON": 0.0, "TURN": ""}

163def climb_out_slots(cifp, profile, ca_point_nm=2.0): 

164 """The runway threshold and ``CA`` slots of a ``climb_out``.""" 

165 climb = profile.climb_out 

166 rwy, bearing_mag, _ = cifp.runway(profile.departure, climb["runway"]) 

167 threshold = _fix_slot(rwy) 

168 ca = _fix_slot(rwy) 

169 ca.update(ID="", PT="CA", CRS=bearing_mag, ALT=_alt_field("+", climb["alt_ft"], "")) 

170 ca["LAT"], ca["LON"] = refnav.direct(rwy.lat, rwy.lon, refnav.wrap360(bearing_mag - profile.magvar_deg), 

171 ca_point_nm) 

172 return [threshold, ca]

175def route_slots(cifp, profile): 

176 """(slots, segments, route name, approach type) for a profile's whole 

177 route: the departure, the en-route items, then the procedure or the 

178 destination.""" 

179 rows = [] # (fix, airway or None) 

180 if profile.departure: 

181 rows.append((cifp.fix(profile.departure, section="PA"), None)) 

182 climb_at = len(rows) 

183 rows += cifp.expand_route(profile.enroute) 

184 proc = profile.procedure 

185 if proc is None: 

186 rows.append((cifp.fix(profile.destination, section="PA"), None)) 

187 slots, segments = [], [] 

188 for r, (f, airway) in enumerate(rows): 

189 if r == climb_at and profile.climb_out: 

190 slots += climb_out_slots(cifp, profile) 

191 segments.append({"KIND": SEGKIND_DEPARTURE, "ID": profile.climb_out["runway"], 

192 "ENT": slots[-2]["ID"], "EXT": slots[-1]["ID"], 

193 "FIRST": len(slots) - 1, "LAST": len(slots)}) 

194 slots[-2]["SGI"] = slots[-1]["SGI"] = len(segments) 

195 slots.append(_fix_slot(f)) 

196 if r == climb_at and profile.climb_out: 

197 slots[-1]["PT"] = "DF" 

198 n = len(slots) 

199 if airway is not None and segments and segments[-1]["KIND"] == SEGKIND_AIRWAY \ 

200 and segments[-1]["ID"] == airway and segments[-1]["LAST"] == n - 1: 

201 segments[-1].update(EXT=f.ident, LAST=n) 

202 else: 

203 segments.append({"KIND": SEGKIND_AIRWAY if airway else SEGKIND_WAYPOINT, "ID": airway or "", 

204 "ENT": f.ident, "EXT": f.ident, "FIRST": n, "LAST": n}) 

205 slots[-1]["SGI"] = len(segments) 

206 names = [profile.departure] if profile.departure else [] 

207 names += [it["fix"] if it["kind"] == "waypoint" else it["id"] for it in profile.enroute] 

208 if proc is None: 

209 names.append(profile.destination) 

210 return slots, segments, " ".join(names)[:32], "" 

211 p_slots, p_segs = procedure_slots(cifp, proc["airport"], proc["approach"], proc["transition"], 

212 profile.magvar_deg) 

213 if slots and slots[-1]["ID"] == p_slots[0]["ID"]: 213 ↛ 214line 213 didn't jump to line 214 because the condition on line 213 was never true

214 raise ValueError(f"the en-route items end at {p_slots[0]['ID']}, where the procedure starts: " 

215 "end them at the fix before it") 

216 off, k0 = len(slots), len(segments) 

217 for s in p_slots: 

218 s["SGI"] += k0 

219 for g in p_segs: 

220 g["FIRST"] += off 

221 g["LAST"] += off 

222 names.append(f"{proc['airport']} {proc['approach']} {proc['transition']}") 

223 return slots + p_slots, segments + p_segs, " ".join(names)[:32], "RNAV"

226def route_block_writes(slots, segments, name, aprtype, cycle, seq): 

227 """The ordered (key, value) writes an editor makes: every slot and 

228 segment row, then the counts, the name and provenance, ``FPLSEQ`` 

229 last (the commit).""" 

230 writes = [] 

231 for i, slot in enumerate(slots, start=1): 

232 for field in ("ID", "LAT", "LON", "TYPE", "PT", "CRS", "DST", "ALT", "SPD", "SGI", 

233 "FLAGS", "CTRLAT", "CTRLON", "TURN"): 

234 writes.append((f"FPL{i}{field}", slot[field])) 

235 for k, seg in enumerate(segments, start=1): 

236 for field in ("KIND", "ID", "ENT", "EXT", "FIRST", "LAST"): 

237 writes.append((f"FPLS{k}{field}", seg[field])) 

238 writes += [("FPLSEGCNT", len(segments)), ("FPLCOUNT", len(slots)), ("FPLNAME", name), 

239 ("FPLAPRTYPE", aprtype), ("FPLDBCYC", cycle), ("FPLSEQ", seq)] 

240 return writes
