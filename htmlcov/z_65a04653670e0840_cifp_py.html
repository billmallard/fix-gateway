3# Minimal ARINC 424-18 reader for the synthetic flight tier (SYN1). 

4# 

5"""Just enough of FAA CIFP (``FAACIFP18``, ARINC 424-18 fixed-width records) 

6to fly one approach and the airways before it: the header cycle, fix 

7positions (terminal and en-route waypoints, VHF navaids, runways, airports), 

8a runway's bearing and elevation, an approach's legs, and an airway's fixes 

9between two of them. 

10 

11Written for the synthetic tier only. It is deliberately not shared with the 

12navigation engine (which reads no CIFP at all -- its route arrives on the 

13bus) nor taken from makerplane-data's ``packtools.arinc424``: the reference 

14navigator must not inherit another decoder's reading of the data. 

15 

16Column positions are 1-based, as ARINC 424-18 section 5 prints them. 

17Anything this module does not understand raises :class:`CifpError`; it 

18never guesses a value the record does not carry. 

19"""

32def _lat(field): 

33 # N/Sddmmsscc -- hundredths of a second 

34 hemi, d, m, s = field[0], int(field[1:3]), int(field[3:5]), int(field[5:9]) / 100.0 

35 v = d + m / 60.0 + s / 3600.0 

36 if hemi not in "NS": 36 ↛ 37line 36 didn't jump to line 37 because the condition on line 36 was never true

37 raise CifpError(f"bad latitude field {field!r}") 

38 return -v if hemi == "S" else v

41def _lon(field): 

42 # E/Wdddmmsscc 

43 hemi, d, m, s = field[0], int(field[1:4]), int(field[4:6]), int(field[6:10]) / 100.0 

44 v = d + m / 60.0 + s / 3600.0 

45 if hemi not in "EW": 45 ↛ 46line 45 didn't jump to line 46 because the condition on line 45 was never true

46 raise CifpError(f"bad longitude field {field!r}") 

47 return -v if hemi == "W" else v

50def _magvar(field): 

51 # E/Wddd.d -> degrees, east positive (ARINC convention; callers convert) 

52 field = field.strip() 

53 if not field: 53 ↛ 54line 53 didn't jump to line 54 because the condition on line 53 was never true

54 return None 

55 hemi, tenths = field[0], int(field[1:5]) 

56 if hemi not in "EWT": 56 ↛ 57line 56 didn't jump to line 57 because the condition on line 56 was never true

57 raise CifpError(f"bad magnetic variation field {field!r}") 

58 return tenths / 10.0 * (-1.0 if hemi == "W" else 1.0)

109class Cifp: 

110 """Decoded subset of one CIFP file. ``cycle`` is the 4-digit AIRAC 

111 cycle from the HDR01 record (``"2611"``).""" 

112 

113 def __init__(self, path): 

114 self.path = str(path) 

115 self.cycle = None 

116 self._lines = [] 

117 with open(path, "r", encoding="ascii", errors="replace") as f: 

118 for raw in f: 

119 line = raw.rstrip("\r\n") 

120 if line.startswith("HDR01"): 

121 # "HDR01FAACIFP18 001P013203973042611 07-OCT-2026..." 

122 self.cycle = _col(line, 36, 39) 

123 elif line.startswith("S"): 123 ↛ 118line 123 didn't jump to line 118 because the condition on line 123 was always true

124 self._lines.append(line) 

125 if not self.cycle or not self.cycle.isdigit(): 125 ↛ 126line 125 didn't jump to line 126 because the condition on line 125 was never true

126 raise CifpError(f"{path}: no HDR01 cycle")

129 def fix(self, ident, airport=None, section=None, region=None): 

130 """The position of ``ident``. ``section`` is the 2-char section code 

131 a leg record names (``PC`` terminal waypoint, ``EA`` en-route 

132 waypoint, ``D `` VHF navaid, ``PG`` runway, ``PA`` airport); with 

133 ``airport`` it disambiguates a terminal waypoint or runway, and 

134 ``region`` (the ICAO code an airway record names, ``K2``) an en-route 

135 waypoint or navaid. Raises if absent or ambiguous.""" 

136 found = [] 

137 for line in self._lines: 

138 if region and _col(line, 5, 5) in "ED" and _col(line, 20, 21) != region: 

139 continue 

140 sec = _col(line, 5, 5) 

141 if sec == "P": 

142 sub = _col(line, 13, 13) 

143 if airport and _col(line, 7, 10).strip() != airport: 

144 continue 

145 if sub == "C" and section in (None, "PC") and _col(line, 14, 18).strip() == ident: 

146 found.append(Fix(ident, _lat(_col(line, 33, 41)), _lon(_col(line, 42, 51)), 

147 "waypoint", _magvar(_col(line, 75, 79)))) 

148 elif sub == "G" and section in (None, "PG") and _col(line, 14, 18).strip() == ident: 

149 found.append(Fix(ident, _lat(_col(line, 33, 41)), _lon(_col(line, 42, 51)), "runway")) 

150 elif sub == "A" and section in (None, "PA") and _col(line, 7, 10).strip() == ident: 

151 found.append(Fix(ident, _lat(_col(line, 33, 41)), _lon(_col(line, 42, 51)), 

152 "airport", _magvar(_col(line, 52, 56)))) 

153 elif sec == "E" and _col(line, 6, 6) == "A" and section in (None, "EA"): 

154 if _col(line, 14, 18).strip() == ident: 

155 found.append(Fix(ident, _lat(_col(line, 33, 41)), _lon(_col(line, 42, 51)), 

156 "waypoint", _magvar(_col(line, 75, 79)))) 

157 elif sec == "D" and _col(line, 6, 6) == " " and section in (None, "D ", "D"): 

158 if _col(line, 14, 17).strip() == ident and _col(line, 33, 33) in "NS": 

159 found.append(Fix(ident, _lat(_col(line, 33, 41)), _lon(_col(line, 42, 51)), 

160 "vor", _magvar(_col(line, 75, 79)))) 

161 if not found: 

162 raise CifpError(f"{ident}: no fix record (section {section!r}, airport {airport!r}) in {self.path}") 

163 if len(set((f.lat, f.lon) for f in found)) > 1: 163 ↛ 164line 163 didn't jump to line 164 because the condition on line 163 was never true

164 raise CifpError(f"{ident}: ambiguous ({len(found)} records); name the section") 

165 return found[0]

167 def runway(self, airport, ident): 

168 """(threshold Fix, magnetic bearing deg, threshold elevation ft) of 

169 runway ``ident`` (``RW30``) at ``airport``, from its ``PG`` record: 

170 bearing cols 28-31 (tenths), elevation cols 67-71. Raises if the 

171 record is absent or either field is blank.""" 

172 for line in self._lines: 

173 if (_col(line, 5, 5) == "P" and _col(line, 13, 13) == "G" 

174 and _col(line, 7, 10).strip() == airport and _col(line, 14, 18).strip() == ident): 

175 brg, elev = _col(line, 28, 31).strip(), _col(line, 67, 71).strip() 

176 if not (brg.isdigit() and elev.lstrip("-").isdigit()): 176 ↛ 177line 176 didn't jump to line 177 because the condition on line 176 was never true

177 raise CifpError(f"{airport} {ident}: runway record without a bearing or an elevation") 

178 f = Fix(ident, _lat(_col(line, 33, 41)), _lon(_col(line, 42, 51)), "runway") 

179 return f, int(brg) / 10.0, float(int(elev)) 

180 raise CifpError(f"{airport} {ident}: no runway record in {self.path}")

182 def enroute_fix(self, ident): 

183 """A fix a pilot enters by name: an en-route waypoint, a VHF navaid 

184 or an airport. Raises if absent, or if two of them with that name 

185 are in different places (name the fix by its airway instead).""" 

186 found = [] 

187 for section in ("EA", "D ", "PA"): 

188 try: 

189 found.append(self.fix(ident, section=section)) 

190 except CifpError as e: 

191 if "ambiguous" in str(e): 191 ↛ 192line 191 didn't jump to line 192 because the condition on line 191 was never true

192 raise 

193 if not found: 193 ↛ 194line 193 didn't jump to line 194 because the condition on line 193 was never true

194 raise CifpError(f"{ident}: no en-route waypoint, navaid or airport record in {self.path}") 

195 if len(set((f.lat, f.lon) for f in found)) > 1: 195 ↛ 196line 195 didn't jump to line 196 because the condition on line 195 was never true

196 raise CifpError(f"{ident}: ambiguous ({', '.join(f.kind for f in found)})") 

197 return found[0]

199 # -- airways ----------------------------------------------------------- 

200 def airway(self, ident, entry, exit): 

201 """The fixes of airway ``ident`` from ``entry`` to ``exit``, both 

202 included, in the order flown: ``[(fix, section, region), ...]``. 

203 

204 An airway is its ``ER`` records in sequence-number order (cols 

205 26-29). The same ident can name more than one airway (the sequence 

206 restarts); the one holding both fixes is used, and more than one 

207 such is an error. Flying it against the record order is the same 

208 fixes reversed.""" 

209 runs, last_seq = [], None 

210 for line in self._lines: 

211 if _col(line, 5, 6) != "ER" or _col(line, 14, 18).strip() != ident: 

212 continue 

213 if _col(line, 39, 39) not in ("0", "1"): 213 ↛ 214line 213 didn't jump to line 214 because the condition on line 213 was never true

214 continue # continuation records name no fix 

215 seq = int(_col(line, 26, 29)) 

216 if last_seq is None or seq <= last_seq: 

217 runs.append([]) 

218 last_seq = seq 

219 runs[-1].append((_col(line, 30, 34).strip(), _col(line, 37, 38), _col(line, 35, 36))) 

220 if not runs: 

221 raise CifpError(f"airway {ident}: no records in {self.path}") 

222 found = [] 

223 for run in runs: 

224 names = [f[0] for f in run] 

225 if entry in names and exit in names: 

226 i, j = names.index(entry), names.index(exit) 

227 found.append(run[i:j + 1] if i <= j else run[j:i + 1][::-1]) 

228 if not found: 

229 raise CifpError(f"airway {ident}: no run through both {entry} and {exit} in {self.path}") 

230 if len(found) > 1: 

231 raise CifpError(f"airway {ident}: {len(found)} runs through {entry} and {exit}") 

232 if len(found[0]) < 2: 232 ↛ 233line 232 didn't jump to line 233 because the condition on line 232 was never true

233 raise CifpError(f"airway {ident}: entry and exit are both {entry}") 

234 return found[0]

236 def expand_route(self, enroute): 

237 """A profile's en-route items (``Profile.enroute``) as the fixes 

238 flown: ``[(Fix, airway), ...]``, where ``airway`` is the airway the 

239 fix belongs to (its entry included), or None for a waypoint entered 

240 on its own. A waypoint straight before an airway is that airway's 

241 entry, as RS1 puts the entry fix in the airway's segment.""" 

242 out = [] 

243 for n, item in enumerate(enroute): 

244 if item["kind"] == "waypoint": 

245 out.append((self.enroute_fix(item["fix"]), None)) 

246 continue 

247 entry = out[-1][0] 

248 fixes = self.airway(item["id"], entry.ident, item["exit"]) 

249 first = self.fix(fixes[0][0], section=fixes[0][1], region=fixes[0][2]) 

250 if (first.lat, first.lon) != (entry.lat, entry.lon): 250 ↛ 251line 250 didn't jump to line 251 because the condition on line 250 was never true

251 raise CifpError(f"airway {item['id']}: its {entry.ident} is not the waypoint entered before it") 

252 out[-1] = (entry, item["id"]) 

253 for ident, section, region in fixes[1:]: 

254 out.append((self.fix(ident, section=section, region=region), item["id"])) 

255 return out
